# P114 validation (iii-b): a size-class rule on the shipped geometry.
# Fruit smaller than 40 mm goes to the right arm's belt (lane 1), everything
# else to the left arm's belt (lane 0). The arm therefore carries a fruit whose
# grade it would never have taken under the shipped grade routing - that is the
# point of the instruction layer.
size<40 => 1
default => 0
