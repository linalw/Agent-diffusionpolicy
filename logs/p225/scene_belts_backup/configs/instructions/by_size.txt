# Size-class sorting (P114 validation iii-b): smaller than 40 mm goes to the
# right belt (destination 1), everything else to the left belt (destination 0).
#
# The carrying arm follows the destination, not the grade: a grade-C strawberry
# ends up on the right and a grade-A kiwi on the left, which the shipped
# grade routing could never do.
size<40 => 1
default => 0
