# The shipped cell's own routing, written as an instruction (P114 validation ii).
# Left belt (destination 0, +Y) takes grade A; right belt (destination 1, -Y)
# takes grade B; grade C is never selected and rides to the belt end.
#
# This is the reference for "the instruction says exactly what the code did":
# with this file armed the run is byte-identical to the un-instructed cell
# (the [sort] rows aside).
grade=A => 0
grade=B => 1
grade=C => skip
